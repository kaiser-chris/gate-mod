gate_expedition_dungeons_base.2 = {
    type = country_event
    dlc = dlc_magic_gate
    placement = scope:dungeon_state

    title = gate_expedition_dungeons_base.2.t
    desc = gate_expedition_dungeons_base.2.d
    flavor = gate_expedition_dungeons_base.2.f

    category = gate_dungeon_preparation

    gui_window = event_window_gate_dungeon_setup

    event_image = {
        video = "middleeast_jungle_expedition"
    }

    immediate = {
        # Create dungeon expedition container
        # and save it as scope expedition (scope:expedition)
        gate_create_dungeon_expedition = {
            dungeon = scope:dungeon
        }
        gate_setup_dungeon_event = yes

        # Create three potential dungeon party leaders
        create_character = {
            template = gate_dungeon_party_leader_template
            save_scope_as = gate_potential_leader
        }
        add_to_variable_list = {
            name = gate_potential_dungeon_party_leaders
            target = scope:gate_potential_leader
        }
        clear_saved_scope = yes
        create_character = {
            template = gate_dungeon_party_leader_template
            save_scope_as = gate_potential_leader
        }
        add_to_variable_list = {
            name = gate_potential_dungeon_party_leaders
            target = scope:gate_potential_leader
        }
        clear_saved_scope = yes
        create_character = {
            template = gate_dungeon_party_leader_template
            save_scope_as = gate_potential_leader
        }
        add_to_variable_list = {
            name = gate_potential_dungeon_party_leaders
            target = scope:gate_potential_leader
        }
        clear_saved_scope = yes
    }

    is_popup = yes

    on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

    icon = "gfx/interface/icons/event_icons/event_gate_expedition.dds"

    option = {
        name = gate_expedition_dungeons_base.2.b
        default_option = yes
        custom_tooltip = {
            text = event_gate_dungeon_cancel_exploration_tooltip
            gate_cleanup_dungeon_expedition = yes
        }
        # Happens in the "after" block
        custom_tooltip = event_gate_dungeon_dismiss_leaders_tooltip
        custom_tooltip_no_bullet = {
            text = event_gate_dungeon_cooldown_tooltip
            set_variable = {
                name = gate_dungeon_delve_cooldown
                days = 180
            }
        }
    }

    option = {
        name = gate_expedition_dungeons_base.2.a
        show_as_unavailable = {
            always = yes
        }
        trigger = {
            custom_tooltip = {
                text = event_gate_dungeon_has_leader_tooltip
                scope:expedition = {
                    var:party = {
                        has_variable = party_leader
                    }
                }
            }
            custom_tooltip = {
                text = event_gate_dungeon_valid_points_tooltip
                scope:expedition = {
                    gate_dungeon_party_points = 0
                }
            }
        }
        highlighted_option = yes
        gate_start_dungeon = {
            expedition = scope:expedition
        }
        hidden_effect = {
            # Remove selected leader from the list so
            # the character isn't killed in the clean up
            remove_list_variable = {
                name = gate_potential_dungeon_party_leaders
                target = scope:expedition.var:party.var:party_leader
            }
        }
    }

    after = {
        hidden_effect = {
            # Clean up unchosen party leaders
            every_in_list = {
                variable = gate_potential_dungeon_party_leaders
                kill_character = {
                    value = yes
                    hidden = yes
                }
            }
            clear_variable_list = gate_potential_dungeon_party_leaders
        }
    }
}
