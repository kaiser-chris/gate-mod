gate_dungeon_delve.1 = {
    type = country_event
    dlc = dlc_magic_gate
    placement = scope:dungeon_state

    title = gate_dungeon_delve_1_t
    desc = gate_dungeon_delve_1_d
    flavor = gate_dungeon_delve_1_f

    gui_window = event_window_gate_dungeon

    event_image = {
        video = "gate_unspecified_dungeon"
    }

    immediate = {
        gate_setup_dungeon_event = yes
    }

    on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

    icon = "gfx/interface/icons/event_icons/event_gate_dungeon.dds"

    duration = 3

    option = {
        name = gate_dungeon_delve_1_a
        highlighted_option = yes
        show_as_unavailable = {
            # Show as disabled instead of
            # hiding the option
            always = yes
        }
        trigger = {
            gate_dungeon_has_renown = {
                amount = 7
            }
        }
        scope:expedition = {
            gate_dungeon_advance_layer = yes
            gate_dungeon_create_log_entry = {
                rating = positive
                event = gate_dungeon_delve_1_t
                outcome_text = gate_dungeon_log_advance
            }
        }
        custom_tooltip_no_bullet = gate_dungeon_tooltip_renown_check
    }

    option = {
        name = gate_dungeon_delve_1_b
        default_option = yes
        random_list = {
            1 = {
                modifier = gate_dungeon_party_cohesion_check_hard_success
                scope:expedition = {
                    gate_dungeon_advance_layer = yes
                    gate_dungeon_party_cohesion_change = {
                        amount = 0.5
                    }
                    gate_dungeon_create_log_entry_value = {
                        rating = positive
                        event = gate_dungeon_delve_1_t
                        outcome_text = gate_dungeon_log_cohesion
                        outcome_value = 0.5
                    }
                    gate_dungeon_create_log_entry = {
                        rating = positive
                        event = gate_dungeon_delve_1_t
                        outcome_text = gate_dungeon_log_advance
                    }
                }
            }
            1 = {
                modifier = gate_dungeon_party_cohesion_check_hard_neutral
                custom_tooltip = gate_dungeon_tooltip_other_route
                gate_dungeon_create_log_entry = {
                    rating = neutral
                    event = gate_dungeon_delve_1_t
                    outcome_text = gate_dungeon_log_stall
                }
            }
            1 = {
                modifier = gate_dungeon_party_cohesion_check_hard_fail
                custom_tooltip = gate_dungeon_tooltip_other_route
                scope:expedition = {
                    gate_dungeon_party_cohesion_change = {
                        amount = -0.5
                    }
                    gate_dungeon_create_log_entry_value = {
                        rating = negative
                        event = gate_dungeon_delve_1_t
                        outcome_text = gate_dungeon_log_cohesion
                        outcome_value = -0.5
                    }
                    gate_dungeon_create_log_entry = {
                        rating = neutral
                        event = gate_dungeon_delve_1_t
                        outcome_text = gate_dungeon_log_stall
                    }
                }
            }
        }
        custom_tooltip_no_bullet = gate_dungeon_tooltip_cohesion_check
    }

    option = {
        name = gate_dungeon_delve_1_c
        random_list = {
            1 = {
                modifier = gate_dungeon_party_condition_check_medium_success
                scope:expedition = {
                    gate_dungeon_advance_layer = yes
                    gate_dungeon_create_log_entry = {
                        rating = positive
                        event = gate_dungeon_delve_1_t
                        outcome_text = gate_dungeon_log_advance
                    }
                }
            }
            1 = {
                modifier = gate_dungeon_party_condition_check_medium_fail
                custom_tooltip = gate_dungeon_tooltip_other_route
                scope:expedition = {
                    gate_dungeon_party_condition_change = {
                        amount = -0.5
                    }
                    gate_dungeon_create_log_entry_value = {
                        rating = negative
                        event = gate_dungeon_delve_1_t
                        outcome_text = gate_dungeon_log_condition
                        outcome_value = -0.5
                    }
                }
            }
        }
        custom_tooltip_no_bullet = gate_dungeon_tooltip_condition_check
    }
}
