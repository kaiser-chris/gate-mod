		name = gate_arcanist_project.1.a

						title = gate_arcanist_project.1.a.fail.t

							amount = -200
						}
						# Also change sgui: gate_arcanist_project.1.a.fail
						# In: common/scripted_guis/gate_message_tooltip_sgui.txt
					}
				}
				60 = {
					# Success
					send_interface_message = {
						title = gate_arcanist_project.1.a.success.t
						desc = gate_arcanist_project.1.a.success.d
						icon = gfx/interface/icons/event_icons/event_gate_project.dds

						gate_add_magic_innovation = {
							amount = 500
						}
						# Also change sgui: gate_arcanist_project.1.a.success
						# In: common/scripted_guis/gate_message_tooltip_sgui.txt
					}
				}
			}
		}
	}

	option = {
		name = gate_arcanist_project.1.b
		default_option = yes
		scope:gate_project_je = {
			add_gate_magic_project_progress = {
				amount = -2
			}
		}

gate_arcanist_project.12 = {
	type = country_event
	dlc = dlc_magic_gate

	title = gate_arcanist_project.12.t
	desc = gate_arcanist_project.12.d
	flavor = gate_arcanist_project.12.f

	event_image = {
		video = "europenorthamerica_london_center"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = gfx/interface/icons/event_icons/event_gate_project.dds

	left_icon = scope:gate_project_researcher

	duration = 3

	cooldown = {
		years = 2
	}

	trigger = {
		gate_project_can_have_generic_event = yes
		var:magic_innovation >= 150
		any_country = {
			is_gate_magic_country = yes
			this != root
			relations:root >= relations_threshold:cordial
		}
	}

	immediate = {
		gate_project_event_save_scopes = yes
		random_country = {
			limit = {
    			is_gate_magic_country = yes
				this != root
				relations:root >= relations_threshold:cordial
			}
			save_scope_as = other_country
		}
	}

	# Trade findings with them
	option = {
		name = gate_arcanist_project.12.a
		scope:gate_project_je = {
			add_gate_magic_project_progress = {
				amount = 4
			}
		}
		gate_add_magic_innovation = {
			amount = -150
		}
		trigger = {
			var:magic_innovation >= 150
		}
		ai_chance = {
			base = 50
		}
	}

	# Our findings stay here
	option = {
		name = gate_arcanist_project.12.b
		default_option = yes
		gate_add_researcher_experience = {
			amount = 3
		}
		ai_chance = {
			base = 50
		}
