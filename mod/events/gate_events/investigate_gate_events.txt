namespace = gate_investigate_gate

gate_investigate_gate.1 = {
	type = country_event
	dlc = dlc_magic_gate
	
	title = gate_investigate_gate.1.t
	desc = gate_investigate_gate.1.d
	flavor = gate_investigate_gate.1.f
	
	event_image = {
		video = "gate_unspecified_magic_gate"
	}

	trigger = {
		any_scope_building = {
			is_building_type = building_gate_alxa
		}
	}

	immediate = {
		set_variable = gate_project_building_gate_alxa_unlocked
		random_scope_building = {
			limit = {
				is_building_type = building_gate_alxa
			}
			state = {
				save_scope_as = gate_state
			}
		}
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_gate_magic_world.dds"
	
	duration = 3

	option = {
		name = gate_investigate_gate.1.a
		default_option = yes
		add_technology_researched = gate_expeditions
	}
}
