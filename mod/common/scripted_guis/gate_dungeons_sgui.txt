sgui_gate_dungeon_layer_complete = {
	scope = container

	saved_scopes = {
		layer
	}

	is_shown = {
		any_in_list = {
			variable = completed_layers
			scope:layer ?= this
		}
	}
}

sgui_gate_change_cohesion = {
	scope = container

	saved_scopes = {
		index
	}

	effect = {
		var:party = {
			set_variable = {
				name = cohesion
				value = {
					value = 1
					subtract = {
						value = 1
						divide = 8
						multiply = scope:index
					}
				}
			}
			clamp_variable = {
				name = cohesion
				min = gate_dungeon_party_cohesion_minimum
				max = gate_dungeon_party_cohesion_maximum
			}
		}
	}
}

gg_project_dungeon = {
	scope = country

	saved_scopes = {
		dungeon
	}

	is_valid = {
		NOT = {
			has_journal_entry = je_gate_expedition_dungeon
		}
		num_pending_events:gate_dungeon_preparation <= 0
	}

	effect = {
		# Start JE
		# Needs scope:dungeon set
		trigger_event = gate_expedition_dungeons_base.2

		# Destroy surrounding containers
		every_in_list = {
			variable = gate_project_dungeon_selector
			destroy_container = yes
		}
		# Clear list
		clear_variable_list = gate_project_dungeon_selector
	}
}

gg_project_dungeon_selector_close = {
	scope = country

	effect = {
		# Destroy surrounding containers
		every_in_list = {
			variable = gate_project_dungeon_selector
			destroy_container = yes
		}
		# Clear list
		clear_variable_list = gate_project_dungeon_selector
	}
}
