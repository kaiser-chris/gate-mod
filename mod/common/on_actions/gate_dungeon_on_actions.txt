on_state_owner_change = {
    on_actions = {
        on_gate_check_dungeon
    }
}

# Reset exploration status after the
# dungeon state owner changed
on_gate_check_dungeon = {
    trigger = {
        any_scope_building = {
            is_building_group = bg_gate_dungeons
        }
    }
    effect = {
        gate_set_dungeon_status = yes
    }
}

on_half_yearly_pulse = {
    on_actions = {
        on_gate_spawn_dungeon
    }
}

# Spawn up to three random dungeons which
# meet the wanted requirements
# (Does nothing if no state can have a new dungeon)
on_gate_spawn_dungeon = {
    effect = {
        random = {
            chance = 20
            gate_spawn_random_dungeon = yes
        }
        random = {
            chance = 20
            gate_spawn_random_dungeon = yes
        }
        random = {
            chance = 20
            gate_spawn_random_dungeon = yes
        }
    }
}

# Triggered via dungeon delve journal entry (monthly)
# root & this = country delving the dungeon
# scope:journal_entry = dungeon delve journal entry
# scope:expedition = expedition script container
on_gate_dungeon_delve_pulse = {
    effect = {
        random_list = {
            30 = {
                # Stall
                # no notification or log entry since nothing changed
            }
            30 = {
                # An event appears
                trigger_event = {
                    on_action = on_gate_dungeon_delve_pulse_event
                }
            }
            40 = {
                # A notification appears
                trigger_event = {
                    on_action = on_gate_dungeon_delve_pulse_notification
                }
                modifier = {
                    value = 1
                    if = {
                        limit = {
                            OR = {
                                scope:expedition.gate_dungeon_party_cohesion_value <= 2
                                scope:expedition.gate_dungeon_party_condition_value <= 2
                            }
                        }
                        # Reduce notifications and therefore increase events
                        # when the delve is close to failure so the player
                        # can feel agency about a potential loss
                        multiply = 0.5
                    }
                }
            }
        }
    }
}

on_gate_dungeon_delve_pulse_notification = {
    effect = {
        random_list = {
            20 = {
                # Stall
                # no notification or log entry since nothing changed
            }
            10 = {
                # Advance
                scope:expedition = {
                    gate_dungeon_advance_layer = yes
                    gate_dungeon_create_log_entry = {
                        rating = positive
                        event = gate_dungeon_log_notification_header
                        outcome_text = gate_dungeon_log_advance
                    }
                }
                modifier = {
                    value = 1
                    if = {
                        limit = {
                            scope:expedition.gate_dungeon_party_condition_percentage > 0.2
                        }
                        add = 1.3
                    }
                    if = {
                        limit = {
                            scope:expedition.gate_dungeon_party_cohesion_percentage < 0.2
                        }
                        multiply = 0.2
                    }
                    else = {
                        multiply = gate_dungeon_party_cohesion_percentage
                    }
                }
            }
            10 = {
                # Increase condition
                gate_dungeon_party_condition_change = {
                    amount = 0.5
                }
                gate_dungeon_create_log_entry_value = {
                    rating = positive
                    event = gate_dungeon_log_notification_header
                    outcome_text = gate_dungeon_log_condition
                    outcome_value = 0.5
                }
                modifier = {
                    value = 1
                    if = {
                        limit = {
                            scope:expedition.gate_dungeon_party_condition_percentage > 0.2
                        }
                        add = 1.3
                    }
                    if = {
                        limit = {
                            scope:expedition.gate_dungeon_party_cohesion_percentage < 0.2
                        }
                        multiply = 0.2
                    }
                    else = {
                        multiply = gate_dungeon_party_cohesion_percentage
                    }
                    if = {
                        limit = {
                            scope:expedition.var:party = {
                                var:condition = var:condition_max
                            }
                        }
                        multiply = 0
                    }
                }
            }
            10 = {
                # Reduce condition
                gate_dungeon_party_condition_change = {
                    amount = -0.5
                }
                gate_dungeon_create_log_entry_value = {
                    rating = negative
                    event = gate_dungeon_log_notification_header
                    outcome_text = gate_dungeon_log_condition
                    outcome_value = -0.5
                }
            }
            10 = {
                # Increase cohesion
                gate_dungeon_party_cohesion_change = {
                    amount = 0.5
                }
                gate_dungeon_create_log_entry_value = {
                    rating = positive
                    event = gate_dungeon_log_notification_header
                    outcome_text = gate_dungeon_log_cohesion
                    outcome_value = 0.5
                }
                modifier = {
                    value = 1
                    if = {
                        limit = {
                            scope:expedition.gate_dungeon_party_condition_percentage > 0.2
                        }
                        add = 1.3
                    }
                    if = {
                        limit = {
                            scope:expedition.gate_dungeon_party_cohesion_percentage < 0.2
                        }
                        multiply = 0.2
                    }
                    else = {
                        multiply = gate_dungeon_party_cohesion_percentage
                    }
                    if = {
                        limit = {
                            scope:expedition.var:party = {
                                var:cohesion = var:cohesion_max
                            }
                        }
                        multiply = 0
                    }
                }
            }
            10 = {
                # Reduce cohesion
                gate_dungeon_party_cohesion_change = {
                    amount = -0.5
                }
                gate_dungeon_create_log_entry_value = {
                    rating = negative
                    event = gate_dungeon_log_notification_header
                    outcome_text = gate_dungeon_log_cohesion
                    outcome_value = -0.5
                }
            }
        }
    }
}
