# Root = Character
on_yearly_pulse_character = {
    on_actions = {
        on_yearly_gate_magic_researcher_experience
    }
}

# Calculate character experience
on_yearly_gate_magic_researcher_experience = {
    trigger = {
        OR = {
            has_role = gate_arcanist
            has_role = gate_active_arcanist
        }
    }
    effect = {
        if = {
            limit = {
                NOT = { has_variable = com_character_background }
            }
            #= Set arcanist character background
            # on_character_spawned_in_pool and on_character_creation do
            # not seem to work
            set_com_character_background = {
                background = character_background_gate_char_role_arcanist
            }
        }
        gate_calculate_researcher_experience = yes
    }
}

on_game_started = {
    on_actions = {
        on_gate_setup_character_background
    }
}

#= Set arcanist character background
# on_character_spawned_in_pool and on_character_creation do
# not seem to work
on_gate_setup_character_background = {
    effect = {
        every_character = {
            limit = {
                OR = {
                    has_role = gate_arcanist
                    has_role = gate_active_arcanist
                }
            }
            set_com_character_background = {
                background = character_background_gate_char_role_arcanist
            }
        }
    }
}

# Root = Character
on_character_death = {
    on_actions = {
        on_gate_magic_researcher_death
    }
}

# Send character death notification
on_gate_magic_researcher_death = {
    trigger = {
        has_role = gate_active_arcanist
    }
    effect = {
        save_scope_as = magic_researcher
        owner = {
            post_notification = message_gate_arcanist_died
        }
    }
}

on_game_started_after_lobby = {
    on_actions = {
        on_game_started_randomize_hiring_dates
    }
}

# Randomize arcanist hiring date
# so they are not 0 at game start
on_game_started_randomize_hiring_dates = {
    effect = {
        every_character = {
            limit = {
                has_variable = gate_hired
            }
            set_variable = {
                name = gate_hired
                value = {
                    value = game_date
                    subtract = { 30 120 }
                }
            }
        }
    }
}
