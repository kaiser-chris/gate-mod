gate_dungeon_party_points_max = 15
gate_dungeon_party_points = {
    value = {
        value = gate_dungeon_party_points_max
        desc = gate_dungeon_party_points_from_base
        format = gate_dungeon_party_points_format
    }
    if = {
        limit = {
            var:party ?= {
                has_variable = party_leader
            }
        }
        add = {
            var:party ?= {
                var:party_leader ?= {
                    value = modifier:character_gate_dungeon_party_points
                }
            }
            desc = gate_dungeon_party_points_from_party_leader
            format = gate_dungeon_party_points_format
        }
    }
    subtract = {
        value = gate_dungeon_party_cohesion_value
        desc = gate_dungeon_party_points_from_cohesion
        format = gate_dungeon_party_points_format
    }
    subtract = {
        value = gate_dungeon_party_condition_value
        desc = gate_dungeon_party_points_from_condition
        format = gate_dungeon_party_points_format
    }
    subtract = {
        value = gate_dungeon_party_renown_value
        desc = gate_dungeon_party_points_from_renown
        format = gate_dungeon_party_points_format
    }
}
