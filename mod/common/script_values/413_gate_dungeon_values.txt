gate_dungeon_party_points_base = 15
gate_dungeon_party_points = {
    value = {
        value = gate_dungeon_party_points_base
        desc = gate_dungeon_party_points_from_base
        format = gate_dungeon_party_points_format
    }
    if = {
        limit = {
            var:party ?= {
                has_variable = party_leader
            }
        }
        add = {
            var:party ?= {
                var:party_leader ?= {
                    value = modifier:character_gate_dungeon_party_points
                }
            }
            desc = gate_dungeon_party_points_from_party_leader
            format = gate_dungeon_party_points_format
        }
    }
    subtract = {
        value = gate_dungeon_party_cohesion_value
        desc = gate_dungeon_party_points_from_cohesion
        format = gate_dungeon_party_points_format
    }
    subtract = {
        value = gate_dungeon_party_condition_value
        desc = gate_dungeon_party_points_from_condition
        format = gate_dungeon_party_points_format
    }
    subtract = {
        value = gate_dungeon_party_renown_value
        desc = gate_dungeon_party_points_from_renown
        format = gate_dungeon_party_points_format
    }
}

gate_dungeon_party_cohesion_minimum = 0
gate_dungeon_party_cohesion_maximum = 8
gate_dungeon_party_cohesion_default = 4
gate_dungeon_party_cohesion_percentage = {
    var:party = {
        if = {
            limit = {
                NOT = {
                    var:cohesion ?= 0
                }
            }
            value = var:cohesion
            divide = 8
        }
        else = {
            value = 0
        }
    }
}
gate_dungeon_party_cohesion_value = {
    var:party = {
        value = var:cohesion
    }
}
gate_dungeon_party_cohesion_max_value = {
    var:party = {
        value = var:cohesion_max
    }
}
gate_dungeon_party_cohesion_check_hard_success = {
    scope:expedition = {
        if = {
            limit = {
                gate_dungeon_party_cohesion_value = 0
            }
            # never succeed when there is no cohesion
            value = 0
        }
        else = {
            value = gate_dungeon_party_cohesion_percentage
            multiply = 100
        }
    }
}
