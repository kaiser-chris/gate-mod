gate_ticks_to_progress = {
    if = {
        limit = {
            scripted_bar_progress:pb_gate_project_tick < 100
        }
        value = 100
        subtract = scripted_bar_progress:pb_gate_project_tick
        owner = {
            divide = gate_project_tick_speed
        }
        round_to = 1
    }
    else = {
        value = 0
    }
}
gate_project_implementation_tick = {
    value = {
        value = gate_project_implementation_tick_base
        desc = pb_gate_project_progress_base
    }
}
gate_project_implementation_tick_base = {
    value = 100
    divide = var:gate_project_cost
}

gate_project_implementation_percent = {
    if = {
        limit = {
            scripted_bar_progress:pb_gate_project_progress > 0
        }
        value = scripted_bar_progress:pb_gate_project_progress
        divide = 100
    }
    else = {
        value = 0
    }
}
