research_magic_decision = {
	is_shown = {
		is_gate_magic_country = no
		NOT = {
			has_journal_entry = je_gate_researching_magic
		}
		NOT = {
			is_country_type = decentralized
		}
	}

	possible = {
		is_gate_magic_country = no
		hidden_trigger = {
			NOT = { has_journal_entry = je_gate_researching_magic }
		}
	}

	when_taken = {
		add_journal_entry = {
			type = je_gate_researching_magic
		}
		trigger_event = {
			id = researching_magic.4
			weeks = 1
		}
		set_variable = {
			name = actively_hiring_magic_researcher
			weeks = 1
		}
	}

	ai_chance = {
		value = 1000
	}
}

decision_gate_start_expedition_reytu = {
	is_shown = {
		AND = {
			AND = {
				has_technology_researched = gate_expeditions
				OR = {
					is_ai = no
					NOT = {
						has_variable = ai_expedition_cooldown
					}
				}
				NOT = {
					has_journal_entry = je_gate_expedition_preparation
				}
				NOT = {
					has_journal_entry = je_gate_expedition_active
				}
			}
			AND = {
				is_player = yes
				has_global_variable = com_debug
			}
		}
	}

	possible = {
		custom_tooltip = {
			text = decision_gate_start_expedition_reytu_possible
			AND = {
				has_building = building_gate_alxa
				AND = {
					is_player = yes
					has_global_variable = com_debug
				}
			}
		}
		NOR = {
			custom_tooltip = {
				text = gate_expedition_blocked_trigger
				has_variable = gate_expedition_blocked
			}
			hidden_trigger = {
				has_variable = gate_expedition
			}
		}
	}

	when_taken = {
		if = {
			limit = {
				is_player = yes
			}
			hidden_effect = {
				create_struct = {
					struct_scope = expedition
					struct_type = gate_expedition_progress
				}
				set_variable = {
					name = gate_expedition
					value = scope:expedition
				}
				var:gate_expedition = {
					set_variable = {
						name = location
						value = global_var:gate_expedition_reytu
					}
				}
			}
			trigger_event = gate_expedition.2
		}
		else = {
			set_variable = AI_EXPEDITION_ANTARCTICA
			ai_expedition_simulator_start = yes
		}
		hidden_effect = {
			set_variable = {
				name = ai_expedition_cooldown
				value = yes
				days = 1825
			}
		}
	}

	ai_chance = {
		value = 0
		if = {
			limit = {
				net_fixed_income > money_amount_multiplier_medium
				gold_reserves > 0
			}
			add = 1
		}
		if = {
			limit = {
				has_strategy = ai_strategy_armed_isolationism
			}
			add = -1000
		}
		if = {
			limit = {
				OR = {
					is_diplomatic_play_committed_participant = yes
					is_at_war = yes
				}
			}
			add = -1000
		}
	}
}

decision_gate_start_dungeon_exploration = {
	is_shown = {
		any_scope_building = {
			is_building_group = bg_gate_dungeons
			has_active_production_method = pm_gate_dungeon_unexplored
		}
	}

	possible = {
		any_scope_building = {
			is_building_group = bg_gate_dungeons
			has_active_production_method = pm_gate_dungeon_unexplored
		}
		NOT = {
			has_journal_entry = je_gate_expedition_dungeon
		}
		custom_tooltip = {
			text = decision_gate_start_dungeon_exploration_is_selector_open_tooltip
			NOT = {
				has_variable_list = gate_project_dungeon_selector
			}
			num_pending_events:gate_dungeon_preparation <= 0
		}
		trigger_if = {
			limit = {
				has_variable = gate_dungeon_delve_cooldown
			}
			custom_tooltip = {
				text = decision_gate_start_dungeon_exploration_cooldown_tooltip
				NOT = {
					has_variable = gate_dungeon_delve_cooldown
				}
			}
		}
	}

	when_taken = {
		custom_tooltip = {
			text = decision_gate_start_dungeon_exploration_selector_tooltip
			every_scope_building = {
				limit = {
					is_building_group = bg_gate_dungeons
					has_active_production_method = pm_gate_dungeon_unexplored
				}
				if = {
					limit = {
						exists = "global_variable_map(dungeons|type)"
					}
					create_container = {
						parent = root
						tags = { gate_mod gate_selector_element }
						save_scope_as = temp_container
					}
					scope:temp_container ?= {
						set_variable = {
							name = building
							value = prev
						}
						set_variable = {
							name = dungeon
							value = "global_variable_map(dungeons|prev.type)"
						}
					}
					root = {
						add_to_variable_list = {
							name = gate_project_dungeon_selector
							target = scope:temp_container
						}
					}
					clear_saved_scope = temp_container
				}
			}
		}
	}

	ai_chance = {
		value = 1000
	}
}
