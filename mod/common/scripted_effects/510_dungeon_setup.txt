gate_setup_dungeon_event = {
	scope:expedition = {
		var:dungeon = {
			var:location = {
				random_scope_state = {
					save_scope_as = dungeon_state
				}
			}
		}
	}
	set_local_variable = {
		name = expedition
		value = scope:expedition
	}
}

gate_create_dungeon_expedition = {
	# Create expedition container
	create_container = {
		tags = { gate_mod gate_expedition }
		save_scope_as = expedition
		parent = this
	}
	# Create party container
	create_container = {
		tags = { gate_mod gate_expedition_party }
		save_scope_as = temp_party
		parent = scope:expedition
	}
	scope:temp_party = {
		set_variable = {
			name = cohesion
			value = gate_dungeon_party_cohesion_default
		}
		set_variable = {
			name = condition
			value = gate_dungeon_party_condition_default
		}
		set_variable = {
			name = renown
			value = gate_dungeon_party_renown_default
		}
	}
	scope:expedition = {
		set_variable = {
			name = party
			value = scope:temp_party
		}
		set_variable = {
			name = dungeon
			value = $dungeon$
		}
		# Set first layer as current step
		var:dungeon = {
			ordered_in_list = {
				variable = layers
				position = 0
				save_scope_as = temp_first_layer
			}
		}
		set_variable = {
			name = current_layer
			value = scope:temp_first_layer
		}
	}
	clear_saved_scope = temp_first_layer
	clear_saved_scope = temp_party
}

gate_create_dungeon = {
	create_container = {
		name = $name$
		tags = { gate_mod gate_dungeon }
		on_created = {
			set_variable = {
				name = name
				value = flag:$name$
			}
			set_variable = {
				name = desc
				value = flag:$name$_desc
			}
			set_variable = {
				name = location
				value = s:$state_region$
			}
			set_variable = {
				name = type
				value = $type$
			}
			set_variable = {
				name = creation_date
				value = game_date
			}
			fix_variable_error = { variable = $name$ }
			fix_variable_error = { variable = $name$_desc }
		}
	}
	add_to_global_variable_list = {
		name = gate_dungeons
		target = container:$name$
	}
}

gate_create_dungeon_layer = {
	create_container = {
		tags = { gate_mod gate_dungeon_layer }
		on_created = {
			set_variable = {
				name = name
				value = flag:$name$
			}
			set_variable = {
				name = desc
				value = flag:$name$_desc
			}
			set_variable = {
				name = type
				value = $type$
			}
			set_variable = is_last
			fix_variable_error = { variable = $name$ }
			fix_variable_error = { variable = $name$_desc }
			fix_variable_error = { variable = gate_dungeon_layers }
		}
		save_scope_as = temp_dungeon_layer
	}
	# Current element is the new last one
	# So remove the designation from all others
	every_in_list = {
		variable = layers
		if = {
			limit = {
				has_variable = is_last
			}
			remove_variable = is_last
		}
	}
	add_to_variable_list = {
		name = layers
		target = scope:temp_dungeon_layer
	}
	scope:temp_dungeon_layer = {
		set_parent = prev
		set_variable = {
			name = index
			value = prev.gate_layer_index_max
		}
	}
	clear_saved_scope = temp_dungeon_layer
}
