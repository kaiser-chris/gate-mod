gate_start_dungeon = {
	custom_tooltip = {
		text = effect_gate_dungeon_start
		add_journal_entry = {
			type = je_gate_expedition_dungeon
			target = $expedition$
		}
		set_local_variable = {
			name = gate_dungeon
			value = $expedition$.var:dungeon
		}
	}
}

gate_cleanup_dungeon_expedition = {
	scope:expedition = {
		var:party = {
			var:party_leader ?= {
				kill_character = {
					value = yes
					hidden = yes
				}
			}
		}
		destroy_container = yes
	}
}

gate_dungeon_advance_layer = {
	hidden_effect = {
		# Complete current layer
		add_to_variable_list = {
			name = completed_layers
			target = var:current_layer
		}
	}

	# Select new current layer
	# (if appropriate)
	if = {
		limit = {
			var:current_layer = {
				has_variable = is_last
			}
		}
		# Finish
		custom_tooltip = {
			text = effect_gate_dungeon_complete
			# We are done
			remove_variable = current_layer

			scope:journal_entry ?= {
				post_notification = message_gate_dungeon_delve_finish
			}
		}
	}
	else = {
		# Advance
		custom_tooltip = {
			text = effect_gate_dungeon_advance
			save_scope_as = temp_expedition

			# Find next layer
			var:dungeon = {
				random_in_list = {
					variable = layers
					limit = {
						# Next layer
						var:index = scope:temp_expedition.gate_layer_index_next
					}
					scope:temp_expedition = {
						set_variable = {
							name = current_layer
							value = prev
						}
					}
				}
			}

			clear_saved_scope = temp_expedition

			scope:journal_entry ?= {
				post_notification = message_gate_dungeon_delve_advance
			}
		}
	}
}

gate_dungeon_party_condition_change = {
	custom_description = {
		text = gate_dungeon_party_condition_change
		value = $amount$
		var:party = {
			change_variable = {
				name = condition
				add = $amount$
			}
			clamp_variable = {
				name = condition
				min = gate_dungeon_party_condition_minimum
				max = gate_dungeon_party_condition_maximum
			}
		}
		scope:journal_entry ?= {
			set_local_variable = {
				name = temp_condition_change
				value = $amount$
			}
			if = {
				limit = {
					$amount$ > 0
				}
				post_notification = message_gate_dungeon_party_condition_increase
			}
			else_if = {
				limit = {
					$amount$ < 0
				}
				post_notification = message_gate_dungeon_party_condition_decrease
			}
		}
	}
}

gate_dungeon_create_log_entry = {
	hidden_effect = {
		create_container = {
			parent = scope:expedition
			save_scope_as = temp_log_entry
			tags = { gate_mod gate_dungeon_log_entry }
		}
		scope:temp_log_entry ?= {
			set_variable = {
				name = rating
				value = flag:gate_dungeon_rating_$rating$
			}
			set_variable = {
				name = event
				value = flag:$event$
			}
			set_variable = {
				name = outcome_text
				value = flag:$outcome_text$
			}
			set_variable = {
				name = creation_date
				value = game_date
			}
		}
		fix_variable_error = { variable = gate_dungeon_rating_$rating$ }
		fix_variable_error = { variable = $event$ }
		fix_variable_error = { variable = $outcome_text$ }
		scope:expedition = {
			add_to_variable_list = {
				name = log_entries
				target = scope:temp_log_entry
			}
		}
		clear_saved_scope = temp_log_entry
	}
}

gate_set_dungeon_building_status = {
	if = {
		limit = {
			has_building = $building$
		}
		if = {
			limit = {
				owner = {
					is_target_in_variable_list = {
						name = gate_completed_dungeons
						target = container:$dungeon$
					}
				}
			}
			activate_production_method = {
				building_type = $building$
				production_method = pm_gate_dungeon_explored
			}
		}
		else = {
			activate_production_method = {
				building_type = $building$
				production_method = pm_gate_dungeon_unexplored
			}
		}
	}
}

gate_build_specific_dungeon = {
	if = {
		limit = {
			# we only need to do something if this is the
			# target state and the building hasn't been built yet
			container:$dungeon$ ?= {
				var:location ?= prev.state_region
				NOT = {
					has_variable = creation_date
				}
			}
			NOT = {
				has_building = $building$
			}
		}
		container:$dungeon$ ?= {
			set_variable = {
				name = creation_date
				value = game_date
			}
		}
		create_building = {
			building = $building$
			level = 1
		}
	}
}
