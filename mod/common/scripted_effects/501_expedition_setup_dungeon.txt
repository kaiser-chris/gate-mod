gate_create_dungeon_expedition = {
	create_container = {
		tags = { gate_expedition }
		on_created = {
			set_variable = {
				name = dungeon
				value = $dungeon$
			}

			# Set first layer as current step
			var:dungeon = {
				ordered_in_list = {
					variable = layers
					position = 0
					save_scope_as = temp_first_layer
				}
			}
			set_variable = {
				name = current_layer
				value = scope:temp_first_layer
			}
		}
		save_scope_as = $scope$
	}
	set_variable = {
		name = expedition
		value = scope:expedition
	}
	clear_saved_scope = temp_first_layer
}

gate_create_dungeon_tyrol = {
	container:gate_dungeon_tyrol ?= {
		destroy_container = yes
	}
	gate_create_dungeon = {
		name = gate_dungeon_tyrol
		state_region = STATE_TYROL
		type = gate_dungeon_cave_01
	}
	container:gate_dungeon_tyrol = {
		gate_create_dungeon_layer = {
			name = gate_dungeon_layer_generic_entrance
			type = gate_dungeon_layer_top_cave_01

		}
		gate_create_dungeon_layer = {
			name = gate_dungeon_layer_generic_one
			type = gate_dungeon_layer_inner_classic_01
			is_last = no
		}
		gate_create_dungeon_layer = {
			name = gate_dungeon_layer_generic_two
			type = gate_dungeon_layer_inner_classic_02
			is_last = no
		}
		gate_create_dungeon_layer = {
			name = gate_dungeon_layer_generic_three
			type = gate_dungeon_layer_inner_classic_03

		}

			name = gate_dungeon_layer_generic_boss
			type = gate_dungeon_layer_inner_classic_04
			is_last = yes
		}
	}
}


gate_create_dungeon = {
	create_container = {
		name = $name$
		tags = { gate_mod gate_dungeon }
		on_created = {
			set_variable = {
				name = name
				value = flag:$name$
			}
			set_variable = {
				name = desc
				value = flag:$name$_desc
			}
			set_variable = {
				name = location
				value = s:$state_region$
			}
			set_variable = {
				name = type
				value = $type$
			}
			fix_variable_error = { variable = $name$ }
			fix_variable_error = { variable = $name$_desc }
		}
	}
	add_to_global_variable_list = {
		name = gate_dungeons
		target = container:$name$
	}
}


gate_create_dungeon_layer = {
	create_container = {
		tags = { gate_mod gate_dungeon_layer }
		on_created = {
			set_variable = {
				name = name
				value = flag:$name$
			}
			set_variable = {
				name = desc
				value = flag:$name$_desc
			}
			set_variable = {
				name = type
				value = $type$
			}
			if = {
				limit = {
					always = $is_last$
				}
				set_variable = is_last
			}
			fix_variable_error = { variable = $name$ }
			fix_variable_error = { variable = $name$_desc }
			fix_variable_error = { variable = gate_dungeon_layers }
		}
		save_scope_as = temp_dungeon_layer
	}
	scope:temp_dungeon_layer = {
		set_parent = prev
	}
	add_to_variable_list = {
		name = layers
		target = scope:temp_dungeon_layer
	}
	clear_saved_scope = temp_dungeon_layer
}
