je_gate_expedition_dungeon = {
	icon = gfx/interface/icons/event_icons/event_gate_dungeon.dds

	group = je_group_gate_dungeons

	widget = {
		gui = "gui/journal_injects/gate_expedition_injects.gui"
		name = "gate_inject_dungeon_layers"
		container = "custom_widget_container_2"
	}

	widget = {
		gui = "gui/com_journal_injects/injects.gui"
		name = "com_journal_entry_dlc_icon"
		container = "custom_widget_container_7"
	}

	immediate = {
		add_gate_dlc_icon = yes
		scope:journal_entry = {
			gate_create_dungeon_expedition = {
				dungeon = container:gate_dungeon_tyrol
				scope = expedition
			}

			random_container = {
				parent = scope:expedition.var:dungeon
				tag = gate_dungeon_layer
				limit = {
					has_variable = is_last
				}
				prev = {
					set_variable = {
						name = last_layer
						value = prev
					}
				}
			}

			# Set active layer
			random_container = {
				parent = container:gate_dungeon_tyrol
				tag = gate_dungeon_layer
				limit = {
					var:name ?= flag:gate_dungeon_layer_generic_two
				}
				scope:expedition = {
					set_variable = {
						name = current_layer
						value = prev
					}
				}
			}

			# Complete first two layers
			every_container = {
				parent = container:gate_dungeon_tyrol
				tag = gate_dungeon_layer
				limit = {
					OR = {
						var:name ?= flag:gate_dungeon_layer_generic_entrance
						var:name ?= flag:gate_dungeon_layer_generic_one
					}
				}
				scope:expedition = {
					add_to_variable_list = {
						name = completed_layers
						target = prev
					}
				}
			}
		}
	}

	complete = {
		custom_tooltip = {
			text = je_gate_expedition_dungeon_complete
			scope:expedition = {
				any_in_list = {
					variable = completed_layers
					scope:journal_entry.var:last_layer ?= this
				}
			}
		}
	}

	transferable = no
	can_revolution_inherit = yes

	weight = 200
	should_be_pinned_by_default_uninvolved_or_context = yes
}
