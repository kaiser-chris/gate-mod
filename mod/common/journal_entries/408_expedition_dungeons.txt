je_gate_expedition_dungeon = {
	icon = "gfx/interface/icons/event_icons/event_gate_expedition.dds"

	group = je_group_expeditions

	widget = {
		gui = "gui/journal_injects/gate_expedition_injects.gui"
		name = "gate_inject_dungeon_layers"
		container = "custom_widget_container_2"
	}

	widget = {
		gui = "gui/com_journal_injects/injects.gui"
		name = "com_journal_entry_dlc_icon"
		container = "custom_widget_container_7"
	}

	immediate = {
		add_gate_dlc_icon = yes
		scope:journal_entry = {
			set_variable = {
				name = dungeon
				value = container:gate_dungeon_tyrol
			}

			# Set active layer
			random_container = {
				parent = container:gate_dungeon_tyrol
				tag = gate_dungeon_layer
				limit = {
					var:name ?= flag:je_gate_expedition_dungeon_step_two
				}
				prev = {
					set_variable = {
						name = current_step
						value = prev
					}
				}
			}

			# Complete first two layers
			every_container = {
				parent = container:gate_dungeon_tyrol
				tag = gate_dungeon_layer
				limit = {
					OR = {
						var:name ?= flag:je_gate_expedition_dungeon_step_supplies
						var:name ?= flag:je_gate_expedition_dungeon_step_one
					}
				}
				prev = {
					add_to_variable_list = {
						name = completed_layers
						target = prev
					}
				}
			}
		}
	}

	complete = {
		scope:journal_entry = {
			var:current_step ?= container:je_gate_expedition_dungeon_step_complete
		}
	}

	transferable = no
	can_revolution_inherit = yes

	weight = 200
	should_be_pinned_by_default_uninvolved_or_context = yes
}
