je_gate_expedition_dungeon = {
	icon = gfx/interface/icons/event_icons/event_gate_dungeon.dds

	group = je_group_gate_dungeons

	widget = {
		gui = "gui/journal_injects/gate_dungeon_injects.gui"
		name = "gate_inject_dungeon"
		container = "custom_widget_container_2"
	}

	widget = {
		gui = "gui/com_journal_injects/injects.gui"
		name = "com_journal_entry_dlc_icon"
		container = "custom_widget_container_7"
	}

	immediate = {
		add_gate_dlc_icon = yes
		debug_log_scopes = yes
		scope:journal_entry = {
			# Create dungeon expedition container
			# and save it as scope expedition
			# (scope:expedition and var:expedition)
			set_variable = {
				name = expedition
				value = scope:target
			}
			scope:target = {
				save_scope_as = expedition
			}

			# Find last layer in dungeon and set it
			# as a variable for the complete condition
			random_container = {
				parent = scope:expedition.var:dungeon
				tag = gate_dungeon_layer
				limit = {
					has_variable = is_last
				}
				prev = {
					set_variable = {
						name = last_layer
						value = prev
					}
				}
			}
		}
	}

	complete = {
		custom_tooltip = {
			text = je_gate_expedition_dungeon_complete
			# Has the last layer been added to the list of completed layers?
			scope:expedition = {
				any_in_list = {
					variable = completed_layers
					scope:journal_entry.var:last_layer ?= this
				}
			}
		}
	}

	on_complete = {
		scope:expedition = {
			# Scope switch for the tooltip
			custom_tooltip = {
				text = je_gate_expedition_dungeon_on_complete

				# Real effect is run in country scope
				root = {
					# Register dungeon as completed
					add_to_variable_list = {
						name = gate_completed_dungeons
						target = scope:target
					}
					# Update production method to unlock
					# the dungeon exploitation production methods
					every_scope_state = {
						gate_set_dungeon_status = yes
					}
				}
			}
		}
		hidden_effect = {
			gate_cleanup_dungeon_expedition = yes
		}
	}

	scripted_button = sb_gate_dungeon_test
	scripted_button = sb_gate_dungeon_advance

	transferable = no
	can_revolution_inherit = yes

	weight = 200
	should_be_pinned_by_default_uninvolved_or_context = yes
}
