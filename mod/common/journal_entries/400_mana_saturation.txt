je_gate_mana_saturation = {
	icon = "gfx/interface/icons/event_icons/event_gate_magic_world.dds"

	group = je_group_global_international_situations
	dlc = dlc_magic_gate

	activation_description = {
		always = yes
	}

	scripted_button = je_gate_mana_saturation_button_slave_ritual

	immediate = {
        set_global_variable = {
            name = global_mana_saturation
            value = 0
        }
		scope:journal_entry = {
			set_variable = com_hide_involved_countries
		}
	}

    should_be_involved = {
        OR = {
            is_player = yes
            NOT = {
                is_country_type = decentralized
            }
        }
    }

    is_shown_when_inactive = {
        hidden_trigger = {
            trigger_if = {
                limit = {
                    exists = global_var:global_mana_saturation
                }
                global_var:global_mana_saturation < 100
            }
            trigger_else = {
                always = no
            }
        }
    }

    possible = {
        hidden_trigger = {
            trigger_if = {
                limit = {
                    exists = global_var:global_mana_saturation
                }
                global_var:global_mana_saturation < 100
            }
            trigger_else = {
                always = no
            }
        }
    }

	map_color = {
		if = {
			limit = {
				has_magic_gate = yes
			}
			# States with gates are marked blue
			value = gate_map_mode_mana_density_gate
		}
		else_if = {
			limit = {
				state_region.var:mana_density > 0
			}
			# States with mana density are marked from red to green
			lerp = {
				factor = {
					value = state_region.var:mana_density
					divide = 100
				}
				min_color = gate_map_mode_mana_density_min
				max_color = gate_map_mode_mana_density_max
			}
		}
	}

	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_gate_mana_saturation_low_progress
				trigger = {
					global_var:global_mana_saturation < 50
				}
			}
			triggered_desc = {
				desc = je_gate_mana_saturation_medium_progress
				trigger = {
					global_var:global_mana_saturation < 80
				}
			}
			triggered_desc = {
				desc = je_gate_mana_saturation_high_progress
				trigger = {
					global_var:global_mana_saturation < 100
				}
			}
		}
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			OR = {
				hidden_trigger = {
					has_global_variable = gate_mana_saturation_full
				}
				hidden_trigger = {
					global_var:global_mana_saturation >= 100
				}
				scope:journal_entry = {
					is_goal_complete = yes
				}
			}
		}
		effect = {
			set_global_variable = {
				name = gate_mana_saturation_full
			}
			set_global_variable = {
				name = global_mana_saturation
				value = 100
			}
		}
		effect_all_involved = {
			trigger_event = {
				id = gate_mana_saturation.1
				popup = yes
			}
		}
	}

	can_revolution_inherit = yes

	current_value = {
		value = global_var:global_mana_saturation
	}

	goal_add_value = {
		value = 100
	}

	progressbar = yes

	weight = 200
	should_be_pinned_by_default_uninvolved_or_context = yes
	should_be_pinned_by_default_involved = yes
}
