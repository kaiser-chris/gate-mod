	icon = "gfx/interface/icons/event_icons/waving_flag.dds"

	group = je_group_foreign_affairs
	dlc = dlc_magic_gate

	activation_description = {
		always = yes
	}

	scripted_button = je_gate_dwarven_alps_integrate_germans
	scripted_button = je_gate_dwarven_alps_button_switzerland_magic
	scripted_button = je_gate_dwarven_alps_button_switzerland_integrate

    is_shown_in_lobby = {
        c:GATE_MK ?= THIS
    }

	immediate = {
		c:SWI ?= {
			save_scope_as = loc_swiss_scope
		}
		c:GATE_MK ?= {
			save_scope_as = loc_raetia_scope
		}
		cu:alemannic ?= {
			save_scope_as = loc_alemannic_scope
		}
		cu:dwarven_south_german ?= {
			save_scope_as = loc_south_german_scope
		}
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			is_at_war = no
			any_scope_state = {
				state_region = s:STATE_TYROL
				is_incorporated = yes
				is_split_state = no
				has_state_trait = state_trait_gate_dwarven_tunnels
			}
			any_scope_state = {
				state_region = s:STATE_SOUTH_TYROL
				is_incorporated = yes
				is_split_state = no
				has_state_trait = state_trait_gate_dwarven_tunnels
			}
			any_scope_state = {
				state_region = s:STATE_EAST_SWITZERLAND
				is_incorporated = yes
				is_split_state = no
				has_state_trait = state_trait_gate_dwarven_tunnels
			}
			any_scope_state = {
				state_region = s:STATE_WEST_SWITZERLAND
				is_incorporated = yes
				is_split_state = no
				has_state_trait = state_trait_gate_dwarven_tunnels
			}
			any_scope_state = {
				state_region = s:STATE_SAVOY
				is_incorporated = yes
				is_split_state = no
				has_state_trait = state_trait_gate_dwarven_tunnels
			}
		}
		effect = {
			trigger_event = {
				id = gate_dwarven_alps.3
				popup = yes
			}
			add_modifier = {
				name = unification_prestige
				days = very_long_modifier_time
			}
		}
	}

	transferable = no
	can_revolution_inherit = yes

	weight = 200
	should_be_pinned_by_default_uninvolved_or_context = yes
}

je_gate_dwarven_integration_of_germans = {
	icon = "gfx/interface/icons/event_icons/waving_flag.dds"

	group = je_group_internal_affairs
	dlc = dlc_magic_gate

	activation_description = {
		always = yes
	}

	immediate = {
		c:SWI ?= {
			save_scope_as = loc_swiss_scope
		}
		c:GATE_MK ?= {
			save_scope_as = loc_raetia_scope
		}
		cu:alemannic ?= {
			save_scope_as = loc_alemannic_scope
		}
		cu:dwarven_south_german ?= {
			save_scope_as = loc_south_german_scope
		}
	}

	on_monthly_pulse = {
		random_events = {
			chance_of_no_event = 95
			2 = gate_dwarven_german_integration.1
			2 = gate_dwarven_german_integration.2
			3 = gate_dwarven_german_integration.3
			3 = gate_dwarven_german_integration.4
			1 = gate_dwarven_german_integration.5
			1 = gate_dwarven_german_integration.6
		}
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			# Complete after the integration progress bar has reached 100
			custom_tooltip = {
				text = je_gate_dwarven_integration_of_germans_success
				scope:journal_entry = {
					scripted_bar_progress = {
						name = pb_gate_german_integration
						value >= 100
					}
				}
			}
		}
		effect = {
			trigger_event = {
				id = gate_dwarven_alps.4
				popup = yes
			}
			add_loyalists = {
				value = very_large_radicals
				culture = cu:dwarven_south_german
			}
			add_loyalists = {
				value = very_large_radicals
				culture = cu:north_german
			}
			add_loyalists = {
				value = very_large_radicals
				culture = cu:alemannic
			}
		}
	}

	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			# Fail when there is no support for the integration
			custom_tooltip = {
				text = je_gate_dwarven_integration_of_germans_failure
				scope:journal_entry = {
					scripted_bar_progress = {
						name = pb_gate_german_integration_support
						value <= 0
					}
				}
			}
		}
	}

	progressbar = no
	scripted_progress_bar = pb_gate_german_integration
	scripted_progress_bar = pb_gate_german_integration_support

	transferable = no
	can_revolution_inherit = yes

	weight = 200
	should_be_pinned_by_default_uninvolved_or_context = yes
}
