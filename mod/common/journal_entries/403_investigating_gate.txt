je_gate_project_investigate_magic_gate_alxa = {
	icon = "gfx/interface/icons/event_icons/event_gate_project.dds"

	group = je_group_gate_projects

	widget = {
		gui = "gui/com_journal_injects/injects.gui"
		name = "com_journal_entry_dlc_icon"
		container = "custom_widget_container_7"
	}

	widget = {
		gui = "gui/com_journal_injects/injects.gui"
		name = "com_journal_entry_building"
		container = "custom_widget_container_1"
	}

	status_desc = {
		triggered_desc = {
			trigger = {
				has_gate_magic_researcher = no
			}
			desc = gg_projects_invalid_arcanist
		}
	}

	immediate = {
		add_gate_dlc_icon = yes
		random_scope_building = {
			limit = {
				is_building_type = building_gate_alxa
			}
			scope:journal_entry = {
				set_variable = {
					name = com_journal_entry_building
					value = prev
				}
			}
		}
	}

	complete = {
		gate_project_check_complete = yes
	}

	on_monthly_pulse = {
		effect = {
			if = {
				limit = {
					any_scope_building = {
						is_building_type = building_gate_alxa
					}
				}
				gate_project_progress = yes
			}
		}
	}

	invalid = {
		OR = {
			hidden_trigger = {
				OR = {
					has_gate_active_magic_project = no
					has_variable = gate_project_building_gate_alxa_unlocked
				}
			}
			NOT = {
				any_scope_building = {
					is_building_type = building_gate_alxa
				}
			}
		}
	}

	on_invalid = {
		hidden_effect = {
			gate_stop_magic_project = yes
		}
	}

	on_complete = {
		trigger_event = {
			id = gate_investigate_gate.1
			popup = yes
		}
		if = {
			limit = {
				has_journal_entry = je_gate_researching_magic
			}
			je:je_gate_researching_magic = {
				add_progress = {
					name = pb_gate_researching_magic
					value = 10
				}
			}
		}
		gate_add_magic_innovation = {
			amount = 500
		}
		gate_add_researcher_experience = {
			amount = 10
		}
		hidden_effect = {
			set_variable = gate_has_finished_project
			set_variable = gate_project_building_gate_alxa_unlocked
			gate_stop_magic_project = yes
		}
	}

	current_value = {
		value = gate_project_current_progress
	}

	goal_add_value = {
		add = scope:target
	}

	scripted_button = gate_project_button_pause
	scripted_button = gate_project_button_continue
	scripted_button = gate_project_button_stop

	display_progressbar_as_months = yes
	progressbar = yes

	weight = 1000

	should_be_pinned_by_default_uninvolved_or_context = yes
}
