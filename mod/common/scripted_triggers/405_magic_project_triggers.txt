gate_project_check_complete = {
    custom_tooltip = {
        text = gate_project_goal
        scope:journal_entry = {
            trigger_if = {
                limit = {
                    has_variable = gate_magic_project_progress
                }
                is_goal_complete = yes
            }
            trigger_else = {
                always = no
            }
        }
    }
}

gate_project_is_paused = {
    NOT = {
        scope:journal_entry = {
            has_modifier = modifier_gate_project_cost
            trigger_if = {
                limit = {
                    var:gate_magic_project_type ?= gate_project_type_infrastructure
                }
                has_modifier = modifier_gate_project_infrastructure_cost
            }
        }
    }
}

gate_project_has_any_selector_open = {
	custom_description = {
		text = gate_project_has_any_selector_open
		OR = {
			has_variable_list = gate_project_improve_mana_extraction_selector
			has_variable_list = gate_project_modernize_seed_selection_selector
		}
	}
}

gate_project_can_start_any = {
	OR = {
		gate_project_can_start_expand_research_institute = yes
		gate_project_can_start_domain = yes
		gate_project_can_start_improve_mana_extraction = yes
		gate_project_can_start_modernize_seed_selection = yes
		gate_project_can_start_investigate_magic_gate = yes
	}
}

gate_project_can_start_expand_research_institute = {
	capital = {
		has_building = building_gate_magic_research_institute
	}
	NAND = {
		has_modifier = modifier_gate_project_institute_new_wing
		has_modifier = modifier_gate_project_institute_improved_instruments
		has_modifier = modifier_gate_project_institute_better_mana_conduits
	}
	is_gate_magic_country = yes
	has_gate_magic_researcher = yes
	has_gate_active_magic_project = no
}

gate_project_can_start_domain = {
	has_technology_researched = gate_formalized_magic
	custom_tooltip = {
		text = je_gate_project_domain_condition_domain_slot
		trigger_if = {
			limit = {
				exists = var:magic_ethics
			}
			var:magic_ethics < 3
		}
		trigger_else = {
			always = no
		}
	}
	is_gate_magic_country = yes
	has_gate_magic_researcher = yes
	has_gate_active_magic_project = no
	custom_tooltip = {
		text = je_gate_project_domain_condition_not_researched
		NOT = {
			has_variable = gate_has_researched_domain
		}
	}
}

gate_project_can_start_change_domain = {
	has_technology_researched = gate_formalized_magic
	is_gate_magic_country = yes
	has_gate_magic_researcher = yes
	has_gate_active_magic_project = no
	custom_tooltip = {
		text = je_gate_project_change_domain_condition_max_domains
		trigger_if = {
			limit = {
				exists = var:magic_ethics
			}
			var:magic_ethics >= 3
		}
		trigger_else = {
			always = no
		}
	}
	custom_tooltip = {
		text = je_gate_project_change_domain_condition_not_researched
		NOT = {
			has_variable = gate_has_researched_change_domains
		}
	}
}
