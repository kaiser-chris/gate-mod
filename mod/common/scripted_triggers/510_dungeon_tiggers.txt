gate_dungeon_has_cohesion = {
	save_temporary_scope_value_as = {
		name = temp_cohesion
		value = $amount$
	}
	custom_tooltip = {
		text = trigger_gate_dungeon_has_cohesion
		subject = $amount$
		scope:expedition = {
			gate_dungeon_party_cohesion_value >= $amount$
		}
	}
}
